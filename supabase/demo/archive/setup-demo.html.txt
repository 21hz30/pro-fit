    <button id="createBtn" onclick="createDemoAccounts()">Create Demo Accounts</button>

  <script type="module">
    import { createClient } from 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2/+esm';

    window.createDemoAccounts = async function() {
      const logEl = document.getElementById('log');
      const btn = document.getElementById('createBtn');
      const serviceKey = document.getElementById('serviceKey').value.trim();

      if (!serviceKey) {
        alert('Please enter your Service Role Key');
        return;
      }

      logEl.style.display = 'block';
      logEl.innerHTML = '';
      btn.disabled = true;

      function log(msg, type = 'info') {
        const line = document.createElement('div');
        line.className = type;
        line.textContent = msg;
        logEl.appendChild(line);
        logEl.scrollTop = logEl.scrollHeight;
      }

      try {
        log('🔧 Initializing Supabase client...');

        const supabase = createClient(
          'https://pntlpqdrdvtsfxneguxt.supabase.co',
          serviceKey,
          {
            auth: {
              autoRefreshToken: false,
              persistSession: false
            }
          }
        );

        const DEMO_ACCOUNTS = [
          {
            email: 'demo-coach@pro-fit.app',
            password: 'ProFit2026!',
            displayName: 'Coach Ben',
            role: 'coach'
          },
          {
            email: 'demo-athlete@pro-fit.app',
            password: 'ProFit2026!',
            displayName: 'Michael',
            role: 'trainee'
          }
        ];

        for (const account of DEMO_ACCOUNTS) {
          log(`\n👤 Creating ${account.displayName} (${account.role})...`);

          // Create auth user
          const { data: authData, error: authError } = await supabase.auth.admin.createUser({
            email: account.email,
            password: account.password,
            email_confirm: true,
            user_metadata: {
              display_name: account.displayName
            }
          });

          if (authError) {
            if (authError.message.includes('already registered')) {
              log(`  ⚠️  User already exists: ${account.email}`, 'info');

              // Get existing user
              const { data: users } = await supabase.auth.admin.listUsers();
              const existingUser = users.users.find(u => u.email === account.email);

              if (existingUser) {
                log(`  🔧 Updating existing user's role...`);
                const { error: profileError } = await supabase
                  .from('profiles')
                  .update({ role: account.role, display_name: account.displayName })
                  .eq('id', existingUser.id);

                if (profileError) {
                  log(`  ❌ Profile update error: ${profileError.message}`, 'error');
                } else {
                  log(`  ✅ Profile updated successfully`, 'success');
                }
              }
              continue;
            }
            log(`  ❌ Auth error: ${authError.message}`, 'error');
            continue;
          }

          log(`  ✅ Auth user created: ${authData.user.id}`, 'success');

          // Update profile role
          const { error: profileError } = await supabase
            .from('profiles')
            .update({ role: account.role, display_name: account.displayName })
            .eq('id', authData.user.id);

          if (profileError) {
            log(`  ❌ Profile error: ${profileError.message}`, 'error');
          } else {
            log(`  ✅ Profile updated with role: ${account.role}`, 'success');
          }

          log(`  📧 Email: ${account.email}`, 'info');
          log(`  🔑 Password: ${account.password}`, 'info');
        }

        log('\n🎉 Demo accounts ready!', 'success');
        log('\nQuick login credentials:', 'info');
        log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━', 'info');
        log('Coach Ben (coach)', 'info');
        log('  Email: demo-coach@pro-fit.app', 'info');
        log('  Password: ProFit2026!', 'info');
        log('', 'info');
        log('Michael (athlete)', 'info');
        log('  Email: demo-athlete@pro-fit.app', 'info');
        log('  Password: ProFit2026!', 'info');

      } catch (error) {
        log(`\n❌ Error: ${error.message}`, 'error');
      } finally {
        btn.disabled = false;
      }
    };
  </script>
